from concurrent.futures import ProcessPoolExecutor

def parallel_structure_compute(molecules, workers=4):
    # Execução paralela otimizada para CPU
    with ProcessPoolExecutor(max_workers=workers) as executor:
        results = list(
            executor.map(compute_molecular_graph, molecules)
        )
    return np.vstack(results)
